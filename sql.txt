Select the web_app database before running the table-creation statement.

CREATE TABLE IF NOT EXISTS Users (
    username VARCHAR(50) PRIMARY KEY,
    password VARCHAR(255) NOT NULL,
    firstname VARCHAR(50) NOT NULL,
    lastname VARCHAR(50) NOT NULL,
    salary DECIMAL(12,2) NOT NULL,
    age INT NOT NULL,
    registerday DATE NOT NULL,
    signintime DATETIME DEFAULT NULL
) DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/*
below are the query templates used by the backend.
nodde supplies the values for each ? placeholder.

1. REGISTER A USER
The password value is a salted scrypt hash.

INSERT INTO Users
    (username, password, firstname, lastname,
     salary, age, registerday, signintime)
VALUES (?, ?, ?, ?, ?, ?, CURDATE(), NULL);


2. FIND AN ACCOUNT FOR SIGN-IN
Node verifies the submitted password against the returned hash

SELECT username, password, firstname, lastname
FROM Users
WHERE username = ?;


RECORD A SUCCESSFUL SIGN-IN
this runs only after password verification succeeds.

UPDATE Users
SET signintime = NOW()
WHERE username = ?;


3. SEARCH BY FIRST NAME

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE firstname LIKE ? ESCAPE '!'
ORDER BY lastname, firstname, username;


SEARCH BY LAST NAME

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE lastname LIKE ? ESCAPE '!'
ORDER BY lastname, firstname, username;


SEARCH BY BOTH NAMES
Both conditions must match.

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE firstname LIKE ? ESCAPE '!'
  AND lastname LIKE ? ESCAPE '!'
ORDER BY lastname, firstname, username;


Search parameters contain surrounding % characters for partial matching
For example, Alex becomes %Alex%.
Literal %, _, and ! characters are escaped by the backend using !.
*/



/* #4 SEARCH BY USERID (aka username)*/

SELECT username, firstname, lastname,
               salary, age, registerday, signintime
        FROM Users
        WHERE username = ?;

/* #5 SEARCH USERS WITH SALARY BETWEEN X AND Y*/
SELECT username, firstname, lastname,
               salary, age, registerday, signintime
        FROM Users
        WHERE salary BETWEEN ? AND ?
        ORDER BY salary, username;

/* #6 SEARCH USERS WITH AGE BETWEEN X AND Y*/
SELECT username, firstname, lastname,
               salary, age, registerday, signintime
        FROM Users
        WHERE age BETWEEN ? AND ?
        ORDER BY age, username;



/* #7 USERS REGISTERED AFTER SPECIFIC USER */
SELECT username, firstname, lastname,
               salary, age, registerday, signintime
        FROM Users
        WHERE registerday > ?
        ORDER BY registerday, username;

/*#8 USERS WHO NEVER SIGNED IN*/
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE signintime IS NULL
ORDER BY registerday, username;


/*#9 USERS WHO REGISTERED ON SAME DAY AS JOHN*/
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday = (
    SELECT registerday FROM Users WHERE username = 'john'
)
AND username <> 'john'
ORDER BY username;
SELECT registerday FROM Users WHERE username = 'john';


/*#10 Users who registered today*/
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users

ORDER BY username;
